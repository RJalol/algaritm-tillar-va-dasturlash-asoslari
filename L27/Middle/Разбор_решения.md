# Разбор решения. L27 · Задание II · №1. Связанный список с вложенным узлом

## Условие
Разработать класс «стек строк» на связанных узлах, объявив узел закрытым вложенным типом. Реализовать добавление, извлечение, счётчик элементов и вывод содержимого. Проверить работу на пяти значениях, включая попытку извлечь из пустого стека.

## Ожидаемый результат
Все операции работают; тип узла недоступен извне; извлечение из пустого стека завершается исключением с понятным сообщением.

## Идея решения
Стек на односвязном списке: вершина указывает на последний добавленный узел. Класс узла закрытый и вложенный — внешний код о нём не знает.

## Решение
```csharp
using System;

class StringStack
{
    private class Node
    {
        public string Value;
        public Node Next;
        public Node(string value, Node next) { Value = value; Next = next; }
    }

    private Node _top;
    public int Count { get; private set; }

    public void Push(string value)
    {
        _top = new Node(value, _top);
        Count++;
    }

    public string Pop()
    {
        if (_top == null)
            throw new InvalidOperationException("Стек пуст: извлекать нечего");
        string v = _top.Value;
        _top = _top.Next;
        Count--;
        return v;
    }

    public void Print()
    {
        for (Node n = _top; n != null; n = n.Next)
            Console.Write(n.Value + " ");
        Console.WriteLine();
    }
}

class Program
{
    static void Main()
    {
        StringStack s = new StringStack();
        foreach (string w in new[] { "один", "два", "три", "четыре", "пять" })
            s.Push(w);

        s.Print();
        Console.WriteLine("Count = " + s.Count);
        Console.WriteLine(s.Pop());

        while (s.Count > 0) s.Pop();
        try { s.Pop(); }
        catch (InvalidOperationException e) { Console.WriteLine(e.Message); }
    }
}
```

## Пошаговое объяснение
1. `Node` — `private` вложенный класс: ссылки на него не выходят за пределы `StringStack`.
2. `Push` создаёт узел, который ссылается на прежнюю вершину, и делает его новой вершиной.
3. `Pop` возвращает значение вершины и сдвигает вершину на следующий узел; для пустого стека бросает `InvalidOperationException` с понятным сообщением.
4. `Count` изменяется только внутри класса (`private set`).
5. Порядок вывода — от вершины: пять, четыре, три, два, один.

## Результат
пять четыре три два один / Count = 5 / пять / Стек пуст: извлекать нечего
