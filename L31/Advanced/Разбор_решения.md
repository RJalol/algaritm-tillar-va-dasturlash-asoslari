# Разбор решения. L31 · Задание III · №1. Простая база заметок

## Условие
Реализовать консольное приложение: добавление, просмотр и удаление заметок с хранением в текстовом файле.

## Ожидаемый результат
Заметки сохраняются между запусками.

## Идея решения
Заметки хранятся в текстовом файле по одной на строку. Консольное меню предлагает добавить, просмотреть и удалить заметку; при каждом изменении файл обновляется, поэтому данные сохраняются между запусками.

## Решение
```csharp
using System;
using System.Collections.Generic;
using System.IO;

class Program
{
    const string FileName = "notes.txt";

    static List<string> Load() =>
        File.Exists(FileName) ? new List<string>(File.ReadAllLines(FileName)) : new List<string>();

    static void Save(List<string> notes) => File.WriteAllLines(FileName, notes);

    static void Main()
    {
        List<string> notes = Load();

        while (true)
        {
            Console.WriteLine("\n1 - показать, 2 - добавить, 3 - удалить, 0 - выход");
            Console.Write("> ");
            switch (Console.ReadLine())
            {
                case "1":
                    if (notes.Count == 0) Console.WriteLine("Заметок нет");
                    for (int i = 0; i < notes.Count; i++)
                        Console.WriteLine((i + 1) + ". " + notes[i]);
                    break;
                case "2":
                    Console.Write("Текст: ");
                    notes.Add(Console.ReadLine());
                    Save(notes);
                    break;
                case "3":
                    Console.Write("Номер: ");
                    if (int.TryParse(Console.ReadLine(), out int n) && n >= 1 && n <= notes.Count)
                    {
                        notes.RemoveAt(n - 1);
                        Save(notes);
                    }
                    else Console.WriteLine("Неверный номер");
                    break;
                case "0":
                    return;
            }
        }
    }
}
```

## Пошаговое объяснение
1. `Load` читает файл в `List<string>` (если файла нет — пустой список): список удобнее массива для добавления и удаления.
2. `Save` целиком перезаписывает файл списком (`File.WriteAllLines`) — простая и надёжная схема для небольших данных.
3. Меню реализовано циклом с `switch`; `int.TryParse` и проверка диапазона защищают от некорректного номера.
4. Сохранение выполняется после каждого изменения, поэтому при следующем запуске программы заметки загружаются из файла.

## Результат
Заметки добавляются, показываются и удаляются; после перезапуска программы они остаются.
