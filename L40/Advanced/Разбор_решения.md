# Разбор решения. L40 · Задание III · №1. Обход лабиринта

## Условие
Найти путь в лабиринте (двумерный массив) поиском в глубину с использованием стека.

## Ожидаемый результат
Путь найден или сообщено об его отсутствии.

## Идея решения
Поиск в глубину: идём по клеткам лабиринта, запоминая в стеке путь. Если тупик — возвращаемся (извлекаем клетку), пока не найдём выход или не опустеет стек.

## Решение
```csharp
using System;
using System.Collections.Generic;

class Program
{
    static void Main()
    {
        string[] maze =
        {
            "S.#.....",
            ".##.###.",
            "....#...",
            "###.#.#.",
            "....#.#E"
        };

        int rows = maze.Length, cols = maze[0].Length;
        bool[,] visited = new bool[rows, cols];
        (int r, int c)[,] parent = new (int, int)[rows, cols];
        Stack<(int r, int c)> stack = new Stack<(int r, int c)>();

        (int r, int c) start = (0, 0), end = (rows - 1, cols - 1);
        stack.Push(start);
        visited[0, 0] = true;
        int[] dr = { 1, -1, 0, 0 }, dc = { 0, 0, 1, -1 };
        bool found = false;

        while (stack.Count > 0 && !found)
        {
            var (r, c) = stack.Pop();
            if ((r, c) == end) { found = true; break; }
            for (int k = 0; k < 4; k++)
            {
                int nr = r + dr[k], nc = c + dc[k];
                if (nr < 0 || nc < 0 || nr >= rows || nc >= cols) continue;
                if (visited[nr, nc] || maze[nr][nc] == '#') continue;
                visited[nr, nc] = true;
                parent[nr, nc] = (r, c);
                stack.Push((nr, nc));
            }
        }

        if (!found) { Console.WriteLine("Пути нет"); return; }

        var path = new List<(int, int)>();
        for (var p = end; p != start; p = parent[p.r, p.c]) path.Add(p);
        path.Add(start);
        path.Reverse();
        Console.WriteLine("Длина найденного пути: " + (path.Count - 1));
        Console.WriteLine(string.Join(" -> ", path));
    }
}
```

## Пошаговое объяснение
1. Лабиринт задан массивом строк: `#` — стена, `.` — проход, `S` — старт (0,0), `E` — выход (правый нижний угол).
2. Стек хранит клетки, которые ещё предстоит обработать; `visited` не даёт вернуться в уже обработанные.
3. Для каждой клетки перебираем четырёх соседей, пропуская выходящих за границы, посещённых и стены; для каждого запоминаем «родителя» — откуда пришли.
4. Когда достигнута клетка выхода, путь восстанавливаем движением по родителям от выхода к старту и разворачиваем список.
5. Если стек опустел, выхода нет — выводится сообщение.
6. Замечание: поиск в глубину находит *какой-то* путь, не обязательно кратчайший (кратчайший гарантирует поиск в ширину на очереди).

## Результат
Выведен найденный путь от S до E и его длина; для лабиринта без выхода — сообщение «Пути нет».
