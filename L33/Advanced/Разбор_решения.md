# Разбор решения. L33 · Задание III · №1. Библиотека геометрических фигур

## Условие
Разработать библиотеку классов (тема №27) с абстрактным классом «фигура» и не менее чем семью производными классами, включая один промежуточный абстрактный («многоугольник»), реализующий часть членов. Абстрактный класс объявляет абстрактные площадь и периметр, реализованный метод описания и виртуальное свойство единицы измерения. Приложение строит массив фигур, сортирует его по площади и печатает таблицу.

## Ожидаемый результат
Таблица построена и отсортирована; промежуточный класс абстрактен и объектом быть не может; ни одна бессмысленная реализация в иерархии не встречается.

## Идея решения
Иерархия: абстрактная `Shape` → абстрактная `Polygon` (реализует периметр по набору сторон, оставляет площадь) → конкретные многоугольники; отдельно — `Circle`, `Ellipse`. Приведён компактный вариант библиотеки и приложения в одном листинге.

## Решение
```csharp
using System;
using System.Linq;

namespace Geometry                                            // библиотека классов
{
    public abstract class Shape
    {
        public abstract double Area { get; }
        public abstract double Perimeter { get; }
        public virtual string Unit => "см";
        public abstract string Name { get; }

        public string Describe() => $"{Name,-14} S = {Area,8:F2} {Unit}²   P = {Perimeter,7:F2} {Unit}";
    }

    public abstract class Polygon : Shape                     // промежуточный абстрактный класс
    {
        protected abstract double[] Sides { get; }
        public override double Perimeter => Sides.Sum();      // реализует часть членов
    }

    public class Rectangle : Polygon
    {
        private readonly double _a, _b;
        public Rectangle(double a, double b) { _a = a; _b = b; }
        public override string Name => "Прямоугольник";
        protected override double[] Sides => new[] { _a, _b, _a, _b };
        public override double Area => _a * _b;
    }

    public class Square : Polygon
    {
        private readonly double _a;
        public Square(double a) { _a = a; }
        public override string Name => "Квадрат";
        protected override double[] Sides => new[] { _a, _a, _a, _a };
        public override double Area => _a * _a;
    }

    public class Triangle : Polygon
    {
        private readonly double _a, _b, _c;
        public Triangle(double a, double b, double c) { _a = a; _b = b; _c = c; }
        public override string Name => "Треугольник";
        protected override double[] Sides => new[] { _a, _b, _c };
        public override double Area
        {
            get { double p = Perimeter / 2; return Math.Sqrt(p * (p - _a) * (p - _b) * (p - _c)); }
        }
    }

    public class RegularPolygon : Polygon
    {
        private readonly int _n; private readonly double _side;
        public RegularPolygon(int n, double side) { _n = n; _side = side; }
        public override string Name => $"{_n}-угольник";
        protected override double[] Sides => Enumerable.Repeat(_side, _n).ToArray();
        public override double Area => _n * _side * _side / (4 * Math.Tan(Math.PI / _n));
    }

    public class Circle : Shape
    {
        private readonly double _r;
        public Circle(double r) { _r = r; }
        public override string Name => "Круг";
        public override double Area => Math.PI * _r * _r;
        public override double Perimeter => 2 * Math.PI * _r;
    }

    public class Ellipse : Shape
    {
        private readonly double _a, _b;
        public Ellipse(double a, double b) { _a = a; _b = b; }
        public override string Name => "Эллипс";
        public override double Area => Math.PI * _a * _b;
        public override double Perimeter =>                      // приближение Рамануджана
            Math.PI * (3 * (_a + _b) - Math.Sqrt((3 * _a + _b) * (_a + 3 * _b)));
    }

    public class Trapezoid : Polygon
    {
        private readonly double _a, _b, _c, _d, _h;
        public Trapezoid(double a, double b, double c, double d, double h) { _a = a; _b = b; _c = c; _d = d; _h = h; }
        public override string Name => "Трапеция";
        protected override double[] Sides => new[] { _a, _b, _c, _d };
        public override double Area => (_a + _c) / 2 * _h;       // основания a и c
    }
}

class Program                                                // приложение
{
    static void Main()
    {
        Geometry.Shape[] shapes =
        {
            new Geometry.Rectangle(3, 4), new Geometry.Square(5), new Geometry.Triangle(3, 4, 5),
            new Geometry.RegularPolygon(6, 2), new Geometry.Circle(2), new Geometry.Ellipse(3, 2),
            new Geometry.Trapezoid(6, 3, 4, 3, 2.5)
        };

        foreach (var s in shapes.OrderBy(s => s.Area))
            Console.WriteLine(s.Describe());

        // new Geometry.Polygon();   // CS0144: промежуточный класс абстрактен
    }
}
```

## Пошаговое объяснение
1. Библиотека — отдельный проект (тема №27); приложение подключает её и использует только открытые типы.
2. `Shape` объявляет абстрактные площадь, периметр и название, реализованный `Describe()` и виртуальную единицу `Unit` (по умолчанию «см», переопределяется при необходимости).
3. `Polygon` — промежуточный абстрактный класс: реализует `Perimeter` как сумму сторон, а `Area` и `Sides` оставляет абстрактными. Поэтому создать «просто многоугольник» нельзя.
4. Семь конкретных классов: прямоугольник, квадрат, треугольник, правильный многоугольник, трапеция (через `Polygon`), круг и эллипс (напрямую от `Shape`).
5. Сортировка по площади выполняется LINQ-ом `OrderBy(s => s.Area)` — ей достаточно общего свойства базового класса.
6. Бессмысленных реализаций нет: каждый член либо реализован осмысленно, либо абстрактен.

## Результат
Таблица из семи фигур, отсортированная по площади; Polygon создать нельзя (CS0144).
